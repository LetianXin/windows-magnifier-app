cmake_minimum_required(VERSION 3.21)
project(EdgeZoom VERSION 0.3.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(EdgeZoom WIN32 src/app.cpp src/renderer.cpp app.rc)
target_include_directories(EdgeZoom PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(EdgeZoom PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(EdgeZoom PRIVATE dwmapi user32 gdi32 shell32)
if(MINGW)
 target_link_options(EdgeZoom PRIVATE -municode -static -s)
endif()
if(MSVC)
 target_compile_options(EdgeZoom PRIVATE /W4 /O2)
endif()
enable_testing()
add_executable(view_test tests/view_test.cpp)
add_test(NAME view_math COMMAND view_test)
